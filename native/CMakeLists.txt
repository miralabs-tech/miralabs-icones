cmake_minimum_required(VERSION 3.16)
project(mira_symbols C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

# -DBUILD_SHARED_LIBS=ON → mira_symbols.dll / .so / .dylib (FFI : Python, C#, Rust, Lur Level 3…)
add_library(mira_symbols mira_symbols.c)
target_include_directories(mira_symbols PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(BUILD_SHARED_LIBS)
  target_compile_definitions(mira_symbols PUBLIC MIRA_SYMBOLS_SHARED)
endif()
if(MSVC)
  target_compile_options(mira_symbols PRIVATE /utf-8 /W3)
else()
  target_compile_options(mira_symbols PRIVATE -Wall -Wextra)
  target_link_libraries(mira_symbols PUBLIC m)
endif()

add_executable(mira_demo demo.c)
target_link_libraries(mira_demo PRIVATE mira_symbols)
